7) Complex API Snippets

Panel + row + binding + validation

val model = object {
  var scriptPath: String = ""
  var verbose: Boolean = false
}

val ui = panel {
  group("Runtime") {
    row("Script path:") {
      textFieldWithBrowseButton("Select script")
        .bindText({ model.scriptPath }, { model.scriptPath = it })
        .trimmedTextValidation(
          DialogValidation.withParameter { textProvider ->
            if (textProvider().isBlank()) error("Path must not be empty") else null
          }
        )
        .align(Align.FILL)
        .resizableColumn()
    }
    row {
      checkBox("Verbose logging")
        .bindSelected({ model.verbose }, { model.verbose = it })
    }
  }
}

Buttons group + typed radio binding

enum class LaunchMode { RUN, DEBUG }
var mode = LaunchMode.RUN

panel {
  buttonsGroup("Launch mode") {
    row { radioButton("Run", LaunchMode.RUN) }
    row { radioButton("Debug", LaunchMode.DEBUG) }
  }.bind({ mode }, { mode = it })
}
